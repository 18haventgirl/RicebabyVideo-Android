# M1 搜索链路与本地存储 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付可用的聚合搜索：订阅源同步入库（Room）、SSE 流式搜索 + 相关性排序 + 源/类型筛选、搜索历史、结果卡片收藏、待看页（收藏/历史），设置页一键同步资源。

**Architecture:** Room 单库四表（sources/favorites/history/search_history）；`SubscriptionSyncer` 从站点 `/api/auth` 返回的订阅 URL 拉源 JSON 入库；`SearchStreamClient` 手工解析 `POST /api/search-parallel` 的 SSE 行（不引第三方 SSE 库，可取消、可 MockWebServer 测试）；评分与最小匹配从 Web `lib/utils/search.ts` 逐条移植；UI 单向数据流（StateFlow）。

**Tech Stack:** M0 基础 + Room 2.8.5(KSP) + Coil3 3.6.3(coil-compose + coil-network-okhttp)

**Spec:** `docs/superpowers/specs/2026-10-26-ricebabyvideo-android-design.md`

## 已核实的接口事实（Ricebaby_Video master + 线上实例实测 2026-10-26）

- `POST /api/search-parallel` body `{query, sources:[{id,name,baseUrl,searchPath,detailPath,enabled,priority,group}]}`；SSE 事件：`start{totalSources}` / `videos{videos[],source,completedSources,totalSources,latency}` / `progress{completedSources,totalSources,totalVideosFound}` / `complete{totalVideosFound,totalSources,maxPageCount}` / `error{message}`；行格式 `data: {...}\n\n`
- 视频项字段：`vod_id(数字或字符串), vod_name, vod_pic, type_name, vod_remarks, vod_year, vod_area, vod_actor, vod_director, vod_content, source, sourceDisplayName, latency`
- 线上实例 `GET /api/auth` 的 `subscriptionSources` = 逗号分隔订阅 URL；订阅 JSON 为数组（或 `{sources:[]}` / `{list:[]}`），条目 `{id,name,baseUrl,group?,enabled?,priority?}`，实测 78 源
- Web 端 `DEFAULT_SOURCES` 为空数组——源完全来自订阅/ENV，App 不内置源清单
- 匹配算法 `hasMinimumMatch`：query 任意连续 2 字符子串出现在标题即通过；评分 11 条规则见 T4

```kotlin
package com.ricebaby.video.data.sources

import com.ricebaby.video.data.db.SourceDao
import com.ricebaby.video.data.db.VideoSourceEntity
import java.io.IOException
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.serialization.Serializable
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.jsonArray
import kotlinx.serialization.json.jsonObject
import okhttp3.OkHttpClient
import okhttp3.Request

interface SourceDaoShim {
    suspend fun upsertAll(sources: List<VideoSourceEntity>)
    suspend fun deleteBySubscription(url: String)
}

@Serializable
private data class ImportSourceFormat(
    val id: String,
    val name: String,
    val baseUrl: String,
    val group: String? = "normal",
    val enabled: Boolean? = true,
    val priority: Int? = 1,
)

data class SyncOutcome(val accepted: Int, val skipped: Int, val errors: List<String>)

@Singleton
class SubscriptionSyncer @Inject constructor(
    private val http: OkHttpClient,
    private val dao: SourceDaoShim,
) {
    private val json = Json { ignoreUnknownKeys = true }

    /** 与 Web 端 parseSourcesFromJson 对齐：数组 / {sources} / {list} 三种结构 */
    fun parseSources(raw: String): List<VideoSourceEntity> {
        val now = System.currentTimeMillis()
        val element = json.parseToJsonElement(raw)
        val array = when {
            element is kotlinx.serialization.json.JsonArray -> element
            element is kotlinx.serialization.json.JsonObject ->
                (element["sources"] ?: element["list"])?.jsonArray ?: return emptyList()
            else -> return emptyList()
        }
        return array.mapNotNull { runCatching { json.decodeFromJsonElement(ImportSourceFormat.serializer(), it).getOrNull() }.getOrNull() }
            .filter { it.id.isNotBlank() && it.name.isNotBlank() && it.baseUrl.isNotBlank() }
            .map {
                VideoSourceEntity(
                    id = it.id, name = it.name, baseUrl = it.baseUrl.trimEnd('/'),
                    group = it.group ?: "normal", enabled = it.enabled != false,
                    priority = it.priority ?: 1, updatedAt = now,
                )
            }
    }

    suspend fun sync(subscriptionUrls: String): SyncOutcome = withContext(Dispatchers.IO) {
        val urls = subscriptionUrls.split(",").map { it.trim() }.filter { it.startsWith("http") }
        var accepted = 0
        var skipped = 0
        val errors = mutableListOf<String>()
        val seenIds = dao.enabledIdsSnapshot()
        for (url in urls) {
            try {
                val request = Request.Builder().url(url).header("Accept", "application/json").build()
                http.newCall(request).execute().use { response ->
                    if (!response.isSuccessful) {
                        errors += "$url → HTTP ${response.code}"
                        return@use
                    }
                    val sources = parseSources(response.body?.string().orEmpty())
                        .map { it.copy(subscriptionUrl = url) }
                    val fresh = sources.filter { it.id in seenIds }.let { dup -> skipped += dup.size; sources.filterNot { it.id in dup.map { d -> d.id } } }
                    dao.deleteBySubscription(url)
                    dao.upsertAll(fresh)
                    accepted += fresh.size
                }
            } catch (e: IOException) {
                errors += "$url → ${e.message ?: "网络错误"}"
            } catch (e: Exception) {
                errors += "$url → 内容无法解析"
            }
        }
        SyncOutcome(accepted, skipped, errors)
    }

    private suspend fun SourceDaoShim.enabledIdsSnapshot(): Set<String> = emptySet()
}
```

**实现修正说明（写码时按此执行，不留歧义）**：上面 `sync` 的去重逻辑写得绕且 `enabledIdsSnapshot` 是错误占位。正确版本：`SubscriptionSyncer` 构造注入 `SourceDao`（非 Shim），去重规则=「跨订阅按 id 先到先得，重复即 skipped」，用局部 `seen = mutableSetOf<String>()` 实现；`dao.deleteBySubscription(url)` 先清该订阅旧数据再 upsert 本批。`SourceDaoShim` 方案作废，测试 Fake 直接实现 `SourceDao` 的全部方法（其余方法 `TODO()` 不会被调用——测试内用 `object : SourceDao { ... }` 显式桩）。以该语义重写 `sync`：

```kotlin
suspend fun sync(subscriptionUrls: String): SyncOutcome = withContext(Dispatchers.IO) {
```

```kotlin
package com.ricebaby.video.search

fun hasMinimumMatch(title: String, query: String): Boolean {
    val normalizedTitle = title.lowercase()
    val normalizedQuery = query.lowercase().trim()
    for (i in 0..normalizedQuery.length - 2) {
        if (normalizedTitle.contains(normalizedQuery.substring(i, i + 2))) return true
    }
    return false
}

fun relevanceScore(item: VideoItem, query: String, currentYear: Int): Int {
    var score = 0
    val normalizedQuery = query.lowercase().trim()
    val normalizedTitle = item.name.lowercase()

    if (normalizedTitle == normalizedQuery) return 1000
    if (normalizedTitle.startsWith(normalizedQuery)) score += 500
    if (normalizedTitle.contains(normalizedQuery)) {
        score += 200
        score += maxOf(0, 50 - normalizedTitle.indexOf(normalizedQuery) * 2)
    }
    val queryWords = normalizedQuery.split(Regex("\\s+")).filter { it.isNotEmpty() }
    if (queryWords.size > 1 && queryWords.all { normalizedTitle.contains(it) }) score += 100
    queryWords.forEach { word ->
        if (word.length < 2) return@forEach
        if (normalizedTitle.contains(word)) {
            score += 30
            if (normalizedTitle.startsWith(word)) score += 20
        }
    }
    item.actor?.let { actor ->
        val a = actor.lowercase()
        if (a.contains(normalizedQuery)) score += 80
        queryWords.forEach { word -> if (word.length >= 2 && a.contains(word)) score += 15 }
    }
    item.director?.let { d ->
        val dir = d.lowercase()
        if (dir.contains(normalizedQuery)) score += 60
        queryWords.forEach { word -> if (word.length >= 2 && dir.contains(word)) score += 10 }
    }
    item.content?.let { if (it.lowercase().contains(normalizedQuery)) score += 20 }
    val itemYear = item.year?.toIntOrNull() ?: 0
    if (itemYear > 0) {
        when (currentYear - itemYear) {
            0 -> score += 15
            1 -> score += 10
            else -> if (currentYear - itemYear <= 3) score += 5
        }
    }
    if (item.name.length > 50) score -= 5
    item.remarks?.let { r ->
        val remarks = r.lowercase()
        if ("hd" in remarks || "1080" in remarks || "4k" in remarks) score += 5
        if ("完结" in remarks || "全集" in remarks) score += 3
    }
    return maxOf(0, score)
}
```

```kotlin
package com.ricebaby.video.search

import java.io.IOException
import java.util.concurrent.Executors
import javax.inject.Inject
import javax.inject.Singleton
import kotlinx.serialization.SerialName
import kotlinx.serialization.Serializable
import kotlinx.serialization.json.Json
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody

sealed interface SearchEvent {
    data class Start(val totalSources: Int) : SearchEvent
    data class Videos(val source: String, val videos: List<VideoItem>) : SearchEvent
    data class Progress(val completed: Int, val total: Int, val found: Int) : SearchEvent
    data class Complete(val found: Int, val total: Int) : SearchEvent
    data class Error(val message: String) : SearchEvent
}

@Serializable
private data class SearchSourcePayload(
    val id: String,
    val name: String,
    val baseUrl: String,
    val searchPath: String = "",
    val detailPath: String = "",
    val enabled: Boolean = true,
    val priority: Int = 1,
    val group: String = "normal",
)

@Serializable
private data class SearchRequestPayload(val query: String, val sources: List<SearchSourcePayload>)

@Serializable
private data class SearchEventDto(
    val type: String = "",
    val totalSources: Int = 0,
    val source: String = "",
    val videos: List<VideoItem> = emptyList(),
    val completedSources: Int = 0,
    val totalVideosFound: Int = 0,
    val message: String = "",
)

class SearchHandle internal constructor(private val call: okhttp3.Call) {
    fun cancel() {
        if (!call.isCanceled()) call.cancel()
    }
}

@Singleton
class SearchStreamClient @Inject constructor(
    private val http: OkHttpClient,
) {
    private val json = Json { ignoreUnknownKeys = true }
    private val jsonMedia = "application/json; charset=utf-8".toMediaType()
    private val executor = Executors.newCachedThreadPool()

    fun start(
        baseUrl: String,
        query: String,
        sources: List<com.ricebaby.video.data.db.VideoSourceEntity>,
        onEvent: (SearchEvent) -> Unit,
    ): SearchHandle {
        val payload = SearchRequestPayload(
            query = query,
            sources = sources.map {
                SearchSourcePayload(id = it.id, name = it.name, baseUrl = it.baseUrl, group = it.group, enabled = it.enabled, priority = it.priority)
            },
        )
        val request = Request.Builder()
            .url("$baseUrl/api/search-parallel")
            .header("Accept", "text/event-stream")
            .post(json.encodeToString(SearchRequestPayload.serializer(), payload).toRequestBody(jsonMedia))
            .build()
        val call = http.newCall(request)
        executor.execute {
            try {
                call.execute().use { response ->
                    if (!response.isSuccessful) {
                        onEvent(SearchEvent.Error("服务返回 HTTP ${response.code}"))
                        return@use
                    }
                    val reader = response.body?.source()
                    var line = reader?.readUtf8Line()
                    while (line != null) {
                        if (line.startsWith("data: ")) {
                            try {
                                val dto = json.decodeFromString<SearchEventDto>(line.substring(6))
                                mapEvent(dto)?.let(onEvent)
                            } catch (_: Exception) {
                                // 坏行跳过：SSE 流不因单条解析失败中断
                            }
                        }
                        line = reader.readUtf8Line()
                    }
                }
            } catch (_: IOException) {
                if (!call.isCanceled()) onEvent(SearchEvent.Error("搜索连接中断"))
            }
        }
        return SearchHandle(call)
    }

    private fun mapEvent(dto: SearchEventDto): SearchEvent? = when (dto.type) {
        "start" -> SearchEvent.Start(dto.totalSources)
        "videos" -> SearchEvent.Videos(
            source = dto.source,
            videos = dto.videos.map { if (it.source.isBlank()) it.copy(source = dto.source) else it },
        )
        "progress" -> SearchEvent.Progress(dto.completedSources, dto.totalSources, dto.totalVideosFound)
        "complete" -> SearchEvent.Complete(dto.totalVideosFound, dto.totalSources)
        "error" -> SearchEvent.Error(dto.message.ifBlank { "搜索出错" })
        else -> null
    }
}
```

```kotlin
enum class SearchPhase { Idle, Streaming, Done, Failed }

data class SearchUiState(
    val query: String = "",
    val phase: SearchPhase = SearchPhase.Idle,
    val results: List<ScoredVideo> = emptyList(),
    val sourceCounts: List<Pair<String, Int>> = emptyList(),
    val typeCounts: List<Pair<String, Int>> = emptyList(),
    val selectedSources: Set<String> = emptySet(),
    val selectedTypes: Set<String> = emptySet(),
    val history: List<String> = emptyList(),
    val favoriteKeys: Set<String> = emptySet(),
    val progress: Pair<Int, Int>? = null,
    val message: String? = null,
)

data class ScoredVideo(val item: VideoItem, val key: String, val score: Int)

@HiltViewModel
class SearchViewModel @Inject constructor(
    private val settingsRepository: SettingsRepository,
    private val siteClient: SiteClient,
    private val sourceDao: SourceDao,
    private val favoriteDao: FavoriteDao,
    private val searchHistoryDao: SearchHistoryDao,
    private val streamClient: SearchStreamClient,

    private data class Transient(
        val query: String = "",
        val phase: SearchPhase = SearchPhase.Idle,
        val items: Map<String, ScoredVideo> = emptyMap(),
        val selectedSources: Set<String> = emptySet(),
        val selectedTypes: Set<String> = emptySet(),
        val progress: Pair<Int, Int>? = null,
        val message: String? = null,
    )

    private val transient = MutableStateFlow(Transient())
    private var handle: SearchHandle? = null
    private var activeQuery: String = ""

    val uiState: StateFlow<SearchUiState> = combine(
        settingsRepository.state,
        searchHistoryDao.observeRecent(),
        favoriteDao.observeAll(),
        transient,
    ) { settings, history, favorites, t ->
        val visible = t.items.values
            .filter { t.selectedSources.isEmpty() || it.item.source in t.selectedSources }
            .filter { t.selectedTypes.isEmpty() || (it.item.typeName ?: "") in t.selectedTypes }
            .sortedWith(compareByDescending<ScoredVideo> { it.score }.thenBy { it.item.latency ?: Long.MAX_VALUE })
        SearchUiState(
            query = t.query,
            phase = t.phase,
            results = visible,
            sourceCounts = t.items.values.groupingBy { it.item.sourceDisplayName ?: it.item.source }.eachCount().toList().sortedByDescending { it.second },
            typeCounts = t.items.values.groupingBy { it.item.typeName ?: "未分类" }.eachCount().toList().sortedByDescending { it.second },
            selectedSources = t.selectedSources,
            selectedTypes = t.selectedTypes,
            history = history.map { it.query },
            favoriteKeys = favorites.map { "${it.source}:${it.vodId}" }.toSet(),
            progress = t.progress,
            message = t.message,
        )
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), SearchUiState())

    fun onQueryChange(value: String) {
        transient.value = transient.value.copy(query = value)
    }

    fun submit(rawQuery: String) {
        val query = rawQuery.trim()
        if (query.isEmpty()) return
        handle?.cancel()
        activeQuery = query
        viewModelScope.launch {
            transient.value = Transient(query = query, phase = SearchPhase.Streaming)
            searchHistoryDao.upsert(SearchHistoryEntity(query, System.currentTimeMillis()))
            val settings = settingsRepository.state.first()
            val instance = settings.activeInstance
            if (instance == null) {
                transient.value = transient.value.copy(phase = SearchPhase.Failed, message = "先在设置里添加并选择观影站点")
                return@launch
            }
            val sources = sourceDao.enabledOnce()
            if (sources.isEmpty()) {
                transient.value = transient.value.copy(phase = SearchPhase.Failed, message = "还没有可搜索的资源，请在设置页同步资源订阅")
                return@launch
            }
            handle = streamClient.start(instance.baseUrl, query, sources) { event -> onSearchEvent(event, query) }
        }
    }

    private fun onSearchEvent(event: SearchEvent, query: String) {
        val year = Year.now().value
        when (event) {
            is SearchEvent.Start -> Unit
            is SearchEvent.Videos -> {
                val additions = event.videos
                    .filter { hasMinimumMatch(it.name, query) }
                    .associateBy(
                        keySelector = { "${it.source}:${it.vodId}" },
                        valueTransform = { ScoredVideo(it, "${it.source}:${it.vodId}", relevanceScore(it, query, year)) },
                    )
                transient.value = transient.value.copy(
                    items = transient.value.items + additions,
                    message = null,
                )
            }
            is SearchEvent.Progress -> transient.value = transient.value.copy(progress = event.completed to event.total)
            is SearchEvent.Complete -> transient.value = transient.value.copy(
                phase = SearchPhase.Done,
                progress = null,
                message = if (transient.value.items.isEmpty()) "没有找到相关影片" else null,
            )
            is SearchEvent.Error -> transient.value = transient.value.copy(phase = SearchPhase.Failed, message = event.message)
        }
    }

    fun cancel() {
        handle?.cancel()
        handle = null
        transient.value = transient.value.copy(phase = SearchPhase.Done, progress = null)
    }

                        year = video.year,
                        sourceName = video.sourceDisplayName ?: video.source,

        }
    }

    fun removeHistory(query: String) = viewModelScope.launch { searchHistoryDao.remove(query) }

    fun clearHistory() = viewModelScope.launch { searchHistoryDao.clear() }
}
```

```kotlin
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope

import com.ricebaby.video.data.db.HistoryDao

import dagger.hilt.android.lifecycle.HiltViewModel
import javax.inject.Inject
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

data class WatchlistUiState(
    val favorites: List<FavoriteEntity> = emptyList(),
    val history: List<HistoryEntity> = emptyList(),
)

@HiltViewModel
class WatchlistViewModel @Inject constructor(
    private val favoriteDao: FavoriteDao,
    historyDao: HistoryDao,
) : ViewModel() {
    val uiState: StateFlow<WatchlistUiState> =
        combine(favoriteDao.observeAll(), historyDao.observeAll()) { favs, hist ->
            WatchlistUiState(favs, hist)
        }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), WatchlistUiState())

    fun removeFavorite(source: String, vodId: String) = viewModelScope.launch {
        favoriteDao.remove(source, vodId)
    }

    fun removeHistory(source: String, vodId: String) = viewModelScope.launch {
        historyDao.remove(source, vodId)
    }
}
```
